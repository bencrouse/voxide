cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
project(voxide_studio_eval VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
option(VOXIDE_OFFLINE "Never download dependencies" ON)
set(VOXIDE_DEPENDENCY_CACHE "${root}/.cache/dependencies" CACHE PATH "Existing pinned archives")
set(VOXIDE_SANITIZER "" CACHE STRING "Diagnostic sanitizer")
if(VOXIDE_SANITIZER)
    if(NOT VOXIDE_SANITIZER MATCHES "^(address|thread)$")
        message(FATAL_ERROR "Unsupported sanitizer")
    endif()
    add_compile_options(-fsanitize=${VOXIDE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${VOXIDE_SANITIZER})
endif()
include("${root}/cmake/JuceDependency.cmake")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
add_subdirectory("${root}/EngineEvaluation" engines EXCLUDE_FROM_ALL)
add_library(eval_session STATIC Session.cpp "${root}/WorkerEvaluation/WorkerEngine.cpp"
    "${root}/InteractionEvaluation/InteractionProcessor.cpp" "${root}/Source/DSP/PitchTracker.cpp"
    "${root}/Source/DSP/HarmonySelector.cpp")
target_include_directories(eval_session PUBLIC . "${root}/EngineEvaluation" "${root}/Source/DSP")
target_compile_definitions(eval_session PRIVATE VOXIDE_WORKER_NO_AUDIT=1)
target_compile_options(eval_session PRIVATE -Wall -Wextra -Wpedantic)
add_library(eval_processor STATIC Processor.cpp Editor.cpp "${root}/EngineEvaluation/EngineAdapter.cpp")
target_include_directories(eval_processor PUBLIC .)
target_compile_definitions(eval_processor PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
    JUCE_USE_FLAC=0 JUCE_USE_OGGVORBIS=0 JUCE_USE_MP3AUDIOFORMAT=0
    JUCE_USE_LAME_AUDIO_FORMAT=0 JUCE_USE_WINDOWS_MEDIA_FORMAT=0)
target_link_libraries(eval_processor PUBLIC eval_session rubberband_comparator juce::juce_audio_utils
    juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
juce_add_plugin(voxide_studio_eval
    COMPANY_NAME "voxide" BUNDLE_ID "com.voxide.audio.studioeval"
    PLUGIN_MANUFACTURER_CODE Vxid PLUGIN_CODE Vx6b FORMATS AU
    PRODUCT_NAME "voxide Studio Eval" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE
    NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE AU_MAIN_TYPE kAudioUnitType_Effect
    AU_SANDBOX_SAFE TRUE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(voxide_studio_eval PRIVATE Entry.cpp)
target_link_libraries(voxide_studio_eval PRIVATE eval_processor)
add_custom_command(TARGET voxide_studio_eval_AU POST_BUILD
    COMMAND /usr/bin/codesign --force --sign - --timestamp=none "$<TARGET_BUNDLE_DIR:voxide_studio_eval_AU>" VERBATIM)
enable_testing()
juce_add_console_app(eval_processor_tests PRODUCT_NAME "eval_processor_tests")
target_sources(eval_processor_tests PRIVATE Tests.cpp)
target_link_libraries(eval_processor_tests PRIVATE eval_processor engine_allocation_audit)
add_test(NAME eval_processor COMMAND eval_processor_tests)
add_library(eval_worker_audited STATIC "${root}/WorkerEvaluation/WorkerEngine.cpp")
target_include_directories(eval_worker_audited PUBLIC "${root}/EngineEvaluation" "${root}/Source/DSP")
target_link_libraries(eval_worker_audited PUBLIC engine_allocation_audit)
add_executable(eval_worker_contract_tests "${root}/WorkerEvaluation/Tests.cpp")
target_link_libraries(eval_worker_contract_tests PRIVATE eval_worker_audited)
add_test(NAME eval_worker_contract COMMAND eval_worker_contract_tests)
if(NOT VOXIDE_SANITIZER)
    add_test(NAME eval_native_audibility COMMAND eval_processor_tests --native)
endif()
add_executable(eval_au_smoke AuSmoke.cpp)
target_link_libraries(eval_au_smoke PRIVATE "-framework AudioToolbox" "-framework CoreFoundation")
add_dependencies(eval_au_smoke voxide_studio_eval_AU)
add_test(NAME eval_au_smoke COMMAND eval_au_smoke "$<TARGET_BUNDLE_DIR:voxide_studio_eval_AU>")
add_test(NAME eval_bundle COMMAND "${Python3_EXECUTABLE}" "${root}/scripts/verify_eval_bundle.py" "$<TARGET_BUNDLE_DIR:voxide_studio_eval_AU>")
