cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
project(voxide_engine_evaluation LANGUAGES C CXX)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64" OR NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "15.0")
    message(FATAL_ERROR "Engine evidence currently supports Apple Silicon/macOS only")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VOXIDE_SANITIZER "" CACHE STRING "address or thread; never use timings from these builds")
if(VOXIDE_SANITIZER)
    if(NOT VOXIDE_SANITIZER MATCHES "^(address|thread)$")
        message(FATAL_ERROR "Unknown sanitizer")
    endif()
    add_compile_options(-fsanitize=${VOXIDE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${VOXIDE_SANITIZER})
endif()
option(VOXIDE_OFFLINE "Never download dependency archives" ON)
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
set(identity_files EngineEvaluation/CMakeLists.txt EngineEvaluation/EngineAdapter.cpp
    EngineEvaluation/EngineAdapter.h EngineEvaluation/NativeEvaluator.cpp
    Source/Tests/AllocationAudit.cpp Source/Tests/AllocationAudit.h cmake/engine-dependencies.json)
list(SORT identity_files)
set(identity "")
foreach(relative IN LISTS identity_files)
    file(SHA256 "${root}/${relative}" hash)
    string(APPEND identity "${relative}|${hash}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/${relative}")
endforeach()
string(SHA256 engine_source_identity "${identity}")
set(offline_arg "")
if(VOXIDE_OFFLINE)
    set(offline_arg --offline)
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${root}/scripts/engine_dependencies.py"
    --destination "${PROJECT_BINARY_DIR}/_deps" ${offline_arg} COMMAND_ERROR_IS_FATAL ANY)
file(READ "${PROJECT_BINARY_DIR}/_deps/engine-sources.json" sources)
foreach(index RANGE 0 2)
    string(JSON name GET "${sources}" packages ${index} name)
    string(JSON source GET "${sources}" packages ${index} source)
    set("source_${name}" "${source}")
    file(GLOB_RECURSE upstream_files CONFIGURE_DEPENDS "${source}/*")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${upstream_files})
endforeach()
# Neither upstream CMake nor Meson may fetch optional dependencies or install tools.
add_library(rubberband_comparator STATIC "${source_rubberband}/single/RubberBandSingle.cpp")
target_include_directories(rubberband_comparator SYSTEM PUBLIC "${source_rubberband}")
target_link_libraries(rubberband_comparator PRIVATE "-framework Accelerate")
add_library(engine_allocation_audit SHARED "${root}/Source/Tests/AllocationAudit.cpp")
if(VOXIDE_SANITIZER)
    target_compile_definitions(engine_allocation_audit PUBLIC VOXIDE_SANITIZED=1)
endif()
enable_testing()
foreach(engine IN ITEMS signalsmith rubberband)
    add_executable(voxide_${engine}_eval NativeEvaluator.cpp EngineAdapter.cpp)
    target_include_directories(voxide_${engine}_eval PRIVATE "${root}/Source/Tests")
    target_compile_options(voxide_${engine}_eval PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(voxide_${engine}_eval PRIVATE VOXIDE_ENGINE_${engine}=1
        VOXIDE_ENGINE_PROFILE="${CMAKE_BUILD_TYPE}" VOXIDE_ENGINE_SANITIZER="${VOXIDE_SANITIZER}")
    target_compile_definitions(voxide_${engine}_eval PRIVATE VOXIDE_ENGINE_SOURCE_IDENTITY="${engine_source_identity}")
    target_compile_definitions(voxide_${engine}_eval PRIVATE VOXIDE_ENGINE_SELFTEST_DIRECTORY="${PROJECT_BINARY_DIR}/test-output")
    if(engine STREQUAL "signalsmith")
        target_include_directories(voxide_${engine}_eval SYSTEM PRIVATE
            "${source_signalsmith-stretch}/include" "${source_signalsmith-linear}/include")
        target_compile_definitions(voxide_${engine}_eval PRIVATE SIGNALSMITH_USE_ACCELERATE=1)
        target_link_libraries(voxide_${engine}_eval PRIVATE "-framework Accelerate")
    else()
        target_link_libraries(voxide_${engine}_eval PRIVATE rubberband_comparator)
    endif()
    target_link_libraries(voxide_${engine}_eval PRIVATE engine_allocation_audit)
    add_test(NAME ${engine}_adapter COMMAND voxide_${engine}_eval --self-test "${PROJECT_BINARY_DIR}/test-output")
    add_test(NAME ${engine}_reset_failure_retention COMMAND "${Python3_EXECUTABLE}"
        "${root}/scripts/engine_reset_retention_test.py" "$<TARGET_FILE:voxide_${engine}_eval>"
        "${PROJECT_BINARY_DIR}/test-output/reset-retention-controls/${engine}")
endforeach()
