cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
project(voxide_allocation_diagnosis LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${root}/EngineEvaluation" engines)
set(files AllocationEvaluation/CMakeLists.txt AllocationEvaluation/Probe.cpp
    ArchitectureEvaluation/TraceAllocations.cpp ArchitectureEvaluation/TraceAllocations.h
    EngineEvaluation/EngineAdapter.cpp EngineEvaluation/EngineAdapter.h
    Source/Tests/AllocationAudit.h cmake/engine-dependencies.json)
list(SORT files)
set(identity "")
foreach(file IN LISTS files)
    file(SHA256 "${root}/${file}" hash)
    string(APPEND identity "${file}|${hash}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/${file}")
endforeach()
string(SHA256 identity "${identity}")
add_library(allocation_diagnosis_trace SHARED "${root}/ArchitectureEvaluation/TraceAllocations.cpp")
add_executable(voxide_rubberband_allocation_probe Probe.cpp "${root}/EngineEvaluation/EngineAdapter.cpp")
target_compile_definitions(voxide_rubberband_allocation_probe PRIVATE
    ALLOCATION_DIAGNOSIS_IDENTITY="${identity}")
target_compile_options(voxide_rubberband_allocation_probe PRIVATE -Wall -Wextra -Wpedantic -g -fno-omit-frame-pointer)
target_link_libraries(voxide_rubberband_allocation_probe PRIVATE rubberband_comparator allocation_diagnosis_trace)
enable_testing()
add_test(NAME allocation_diagnosis_positive_control COMMAND voxide_rubberband_allocation_probe --self-test)
