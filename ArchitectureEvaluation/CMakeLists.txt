cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
project(voxide_architecture_research LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VOXIDE_SANITIZER "" CACHE STRING "address or thread")
if(VOXIDE_SANITIZER)
    if(NOT VOXIDE_SANITIZER MATCHES "^(address|thread)$")
        message(FATAL_ERROR "Unsupported sanitizer")
    endif()
    add_compile_options(-fsanitize=${VOXIDE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${VOXIDE_SANITIZER})
endif()
enable_testing()
add_subdirectory(../EngineEvaluation engines)
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
file(READ "${PROJECT_BINARY_DIR}/engines/_deps/engine-sources.json" dependencies)
foreach(index RANGE 0 2)
    string(JSON name GET "${dependencies}" packages ${index} name)
    string(JSON source GET "${dependencies}" packages ${index} source)
    set("source_${name}" "${source}")
endforeach()
set(files ArchitectureEvaluation/CMakeLists.txt ArchitectureEvaluation/Probe.cpp
    ArchitectureEvaluation/SignalsmithAdapter.cpp ArchitectureEvaluation/TraceAllocations.cpp
    ArchitectureEvaluation/TraceAllocations.h EngineEvaluation/EngineAdapter.cpp EngineEvaluation/EngineAdapter.h
    Source/DSP/PitchTracker.cpp Source/DSP/PitchTracker.h Source/Tests/AllocationAudit.cpp
    Source/Tests/AllocationAudit.h Source/DSP/HarmonySelector.cpp Source/DSP/HarmonySelector.h cmake/engine-dependencies.json)
list(SORT files)
set(identity "")
foreach(file IN LISTS files)
    file(SHA256 "${root}/${file}" hash)
    string(APPEND identity "${file}|${hash}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/${file}")
endforeach()
string(SHA256 identity "${identity}")
add_library(architecture_trace SHARED TraceAllocations.cpp)
if(VOXIDE_SANITIZER)
    target_compile_definitions(architecture_trace PUBLIC VOXIDE_SANITIZED=1)
endif()
foreach(backend IN ITEMS accelerate builtin rubberband)
    if(backend STREQUAL "rubberband")
        set(adapter "${root}/EngineEvaluation/EngineAdapter.cpp")
    else()
        set(adapter SignalsmithAdapter.cpp)
    endif()
    add_executable(architecture_${backend} Probe.cpp ${adapter} "${root}/Source/DSP/PitchTracker.cpp" "${root}/Source/DSP/HarmonySelector.cpp")
    target_include_directories(architecture_${backend} PRIVATE "${root}/EngineEvaluation" "${root}/Source/DSP" "${root}/Source/Tests")
    target_compile_options(architecture_${backend} PRIVATE -Wall -Wextra -Wpedantic -fno-omit-frame-pointer -g)
    target_compile_definitions(architecture_${backend} PRIVATE ARCHITECTURE_IDENTITY="${identity}"
        ARCHITECTURE_BUILD="${CMAKE_BUILD_TYPE}" ARCHITECTURE_SANITIZER="${VOXIDE_SANITIZER}" ARCHITECTURE_BACKEND="${backend}")
    target_link_libraries(architecture_${backend} PRIVATE architecture_trace)
    if(backend STREQUAL "rubberband")
        target_link_libraries(architecture_${backend} PRIVATE rubberband_comparator)
    else()
        target_include_directories(architecture_${backend} SYSTEM PRIVATE "${source_signalsmith-stretch}/include" "${source_signalsmith-linear}/include")
        if(backend STREQUAL "accelerate")
            target_compile_definitions(architecture_${backend} PRIVATE SIGNALSMITH_USE_ACCELERATE=1)
            target_link_libraries(architecture_${backend} PRIVATE "-framework Accelerate")
        endif()
    endif()
    add_test(NAME architecture_${backend}_contracts COMMAND architecture_${backend} --self-test)
endforeach()
