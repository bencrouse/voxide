cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
project(voxide_worker_feasibility LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VOXIDE_SANITIZER "" CACHE STRING "Diagnostic sanitizer")
if(VOXIDE_SANITIZER)
    if(NOT VOXIDE_SANITIZER MATCHES "^(address|thread)$")
        message(FATAL_ERROR "Unsupported sanitizer")
    endif()
    add_compile_options(-fsanitize=${VOXIDE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${VOXIDE_SANITIZER})
endif()
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
add_subdirectory("${root}/InteractionEvaluation" interaction)
set(files WorkerEvaluation/CMakeLists.txt WorkerEvaluation/WorkerEngine.h WorkerEvaluation/WorkerEngine.cpp
    WorkerEvaluation/Tests.cpp WorkerEvaluation/Probe.cpp InteractionEvaluation/InteractionProcessor.cpp
    InteractionEvaluation/InteractionProcessor.h EngineEvaluation/EngineAdapter.cpp EngineEvaluation/EngineAdapter.h
    Source/DSP/PitchTracker.cpp Source/DSP/PitchTracker.h Source/DSP/HarmonySelector.cpp Source/DSP/HarmonySelector.h
    Source/Tests/AllocationAudit.cpp Source/Tests/AllocationAudit.h cmake/engine-dependencies.json)
list(SORT files)
set(identity "")
foreach(file IN LISTS files)
    file(SHA256 "${root}/${file}" hash)
    string(APPEND identity "${file}|${hash}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/${file}")
endforeach()
string(SHA256 identity "${identity}")
add_library(worker_core STATIC WorkerEngine.cpp)
target_link_libraries(worker_core PUBLIC interaction_core engine_allocation_audit)
target_compile_options(worker_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(worker_contract_tests Tests.cpp)
target_link_libraries(worker_contract_tests PRIVATE worker_core)
add_executable(voxide_worker_probe Probe.cpp "${root}/EngineEvaluation/EngineAdapter.cpp")
target_link_libraries(voxide_worker_probe PRIVATE worker_core rubberband_comparator)
target_compile_definitions(voxide_worker_probe PRIVATE WORKER_IDENTITY="${identity}"
    WORKER_PROFILE="${CMAKE_BUILD_TYPE}" WORKER_SANITIZER="${VOXIDE_SANITIZER}")
target_compile_options(voxide_worker_probe PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_test(NAME worker_contract COMMAND worker_contract_tests)
