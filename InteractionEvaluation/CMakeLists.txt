cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Supported architecture")
set(CMAKE_OSX_DEPLOYMENT_TARGET "15.0" CACHE STRING "Minimum macOS")
project(voxide_interaction_evaluation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(VOXIDE_SANITIZER "" CACHE STRING "Diagnostic sanitizer")
if(VOXIDE_SANITIZER)
    if(NOT VOXIDE_SANITIZER MATCHES "^(address|thread)$")
        message(FATAL_ERROR "Unknown sanitizer")
    endif()
    add_compile_options(-fsanitize=${VOXIDE_SANITIZER} -fno-omit-frame-pointer)
    add_link_options(-fsanitize=${VOXIDE_SANITIZER})
endif()
set(root "${CMAKE_CURRENT_LIST_DIR}/..")
find_package(Python3 3.9 REQUIRED COMPONENTS Interpreter)
enable_testing()
# Reuse PR04's exact pins/builds/auditor, never the product AU target.
add_subdirectory("${root}/EngineEvaluation" engines)
file(READ "${PROJECT_BINARY_DIR}/engines/_deps/engine-sources.json" dependencies)
foreach(index RANGE 0 2)
    string(JSON name GET "${dependencies}" packages ${index} name)
    string(JSON source GET "${dependencies}" packages ${index} source)
    set("source_${name}" "${source}")
endforeach()
set(identity_files InteractionEvaluation/CMakeLists.txt InteractionEvaluation/InteractionProcessor.cpp
    InteractionEvaluation/InteractionProcessor.h InteractionEvaluation/InteractionProbe.cpp InteractionEvaluation/InteractionTests.cpp
    InteractionEvaluation/EngineeringProbe.cpp InteractionEvaluation/EngineeringProbe.h
    Source/DSP/PitchTracker.cpp Source/DSP/PitchTracker.h Source/DSP/HarmonySelector.cpp Source/DSP/HarmonySelector.h
    EngineEvaluation/EngineAdapter.cpp EngineEvaluation/EngineAdapter.h Source/Tests/AllocationAudit.cpp Source/Tests/AllocationAudit.h
    cmake/engine-dependencies.json)
list(SORT identity_files)
set(identity "")
foreach(relative IN LISTS identity_files)
    file(SHA256 "${root}/${relative}" hash)
    string(APPEND identity "${relative}|${hash}\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${root}/${relative}")
endforeach()
string(SHA256 source_identity "${identity}")
add_library(interaction_core STATIC InteractionProcessor.cpp "${root}/Source/DSP/PitchTracker.cpp" "${root}/Source/DSP/HarmonySelector.cpp")
target_include_directories(interaction_core PUBLIC . "${root}/Source/DSP" "${root}/EngineEvaluation")
target_compile_options(interaction_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(interaction_contract_tests InteractionTests.cpp)
target_include_directories(interaction_contract_tests PRIVATE "${root}/Source/Tests")
target_link_libraries(interaction_contract_tests PRIVATE interaction_core engine_allocation_audit)
add_test(NAME interaction_contract COMMAND interaction_contract_tests)
foreach(engine IN ITEMS signalsmith rubberband)
    add_executable(voxide_${engine}_interaction InteractionProbe.cpp EngineeringProbe.cpp "${root}/EngineEvaluation/EngineAdapter.cpp")
    target_include_directories(voxide_${engine}_interaction PRIVATE "${root}/Source/Tests")
    target_compile_options(voxide_${engine}_interaction PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_definitions(voxide_${engine}_interaction PRIVATE VOXIDE_ENGINE_${engine}=1
        VOXIDE_INTERACTION_PROFILE="${CMAKE_BUILD_TYPE}" VOXIDE_INTERACTION_SANITIZER="${VOXIDE_SANITIZER}"
        VOXIDE_INTERACTION_SOURCE_IDENTITY="${source_identity}")
    target_link_libraries(voxide_${engine}_interaction PRIVATE interaction_core engine_allocation_audit)
    if(engine STREQUAL "signalsmith")
        target_include_directories(voxide_${engine}_interaction SYSTEM PRIVATE "${source_signalsmith-stretch}/include" "${source_signalsmith-linear}/include")
        target_compile_definitions(voxide_${engine}_interaction PRIVATE SIGNALSMITH_USE_ACCELERATE=1)
        target_link_libraries(voxide_${engine}_interaction PRIVATE "-framework Accelerate")
    else()
        target_link_libraries(voxide_${engine}_interaction PRIVATE rubberband_comparator)
    endif()
    add_test(NAME ${engine}_interaction_smoke COMMAND voxide_${engine}_interaction --self-test)
endforeach()
